    42→/* ============ 位移球体（与 surf() 位移场严格一致, 供脂质/NPC 对齐表面） ============ */
    43→
    44→function displacedSphere(R: number, detail: number, freq: number, amp: number, seed: number): THREE.BufferGeometry {
    45→  const geo = new THREE.IcosahedronGeometry(R, detail);
    46→  const pos = geo.attributes.position as THREE.BufferAttribute;
    47→  const v = new THREE.Vector3();
    48→  for (let i = 0; i < pos.count; i++) {
    49→    v.fromBufferAttribute(pos, i).normalize();
    50→    const d = (fbm3(v.x * freq, v.y * freq, v.z * freq, 3, seed) - 0.5) * 2 * amp;
    51→    v.multiplyScalar(R + d);
    52→    pos.setXYZ(i, v.x, v.y, v.z);
    53→  }
    54→  pos.needsUpdate = true;
    55→  geo.computeVertexNormals();
    56→  return geo;
    57→}
    58→
    59→/** 方向 dir 处的表面半径（位移场） */
    60→function surf(dir: THREE.Vector3, R: number, freq: number, amp: number, seed: number): number {
    61→  return R + (fbm3(dir.x * freq, dir.y * freq, dir.z * freq, 3, seed) - 0.5) * 2 * amp;
    62→}

    47→/** 低端设备/软件渲染检测（SwiftShader/CPU 渲染/低核数 → 自动流畅模式，降帧缓冲内存与 CPU 负担）
    48→ *  模块级一次性缓存: 在 Canvas 创建前完成探测，保证首帧即使用正确的渲染参数 */
    49→let _lowEndCache: boolean | null = null;
    50→function detectLowEndGpu(): boolean {
    51→  if (typeof navigator === 'undefined') return false;
    52→  if (_lowEndCache !== null) return _lowEndCache;
    53→  const nav = navigator as Navigator & { deviceMemory?: number };
    54→  let low = false;
    55→  if (nav.deviceMemory !== undefined && nav.deviceMemory <= 4) low = true;
    56→  else if (nav.hardwareConcurrency && nav.hardwareConcurrency <= 4) low = true;
    57→  if (!low) {
    58→    try {
    59→      const c = document.createElement('canvas');
    60→      const gl = (c.getContext('webgl2') ?? c.getContext('webgl')) as WebGLRenderingContext | null;
    61→      if (!gl) low = true;
    62→      else {
    63→        const ext = gl.getExtension('WEBGL_debug_renderer_info');
    64→        const renderer = ext ? String(gl.getParameter(ext.UNMASKED_RENDERER_WEBGL)) : '';
    65→        low = /swiftshader|llvmpipe|software|basic\s*render|angle \(.*software/i.test(renderer);
    66→      }
    67→    } catch {
    68→      low = false;
    69→    }
    70→  }
    71→  _lowEndCache = low;
    72→  return low;
    73→}

    94→/** 相机驱动器: 预设机位（含标准观察方位） + 信号跟随 + 教学聚焦（阻尼插值）
    95→ *  自由交互保障: 模式切换仅在前 2s 过渡期内收敛机位; 用户拖拽/滚轮后 4s 内完全让位
    96→ *  （OrbitControls 全权接管, 含 autoRotate） —— 之后的旋转/缩放不再被对抗 */
    97→function CameraRig({ mode, layout, spec, controlsRef, tourTarget }: {
    98→  mode: CamMode;
    99→  layout: ReturnType<typeof layout3D> | null;
   100→  spec: { scale: [number, number, number] };
   101→  controlsRef: RefObject<OrbitControlsImpl | null>;
   102→  /** 教学引导: 当前聚焦分子世界坐标 */
   103→  tourTarget: Vec3 | null;
   104→}) {
   105→  const { camera, gl } = useThree();
   106→  const desired = useRef({ target: new THREE.Vector3(), dist: 30, dir: new THREE.Vector3(0, 0.33, 0.94) });
   107→  const smoothTarget = useRef(new THREE.Vector3());
   108→  const lastTracked = useRef<string | null>(null);
   109→  /** 模式切换时间戳（过渡期内才收敛机位） */
   110→  const modeSince = useRef(0);
   111→  /** 用户最近一次交互（拖拽/滚轮/触摸）时间戳 */
   112→  const lastUser = useRef(-1e9);
   113→
   114→  const toWorld = (p: Vec3): THREE.Vector3 =>
   115→    new THREE.Vector3(p.x * spec.scale[0], p.y * spec.scale[1], p.z * spec.scale[2]);
   116→
   117→  // 用户交互检测（直接监听画布事件, 不依赖 controls 实例时序）
   118→  useEffect(() => {
   119→    const el = gl.domElement;
   120→    const onInteract = () => { lastUser.current = performance.now(); };
   121→    el.addEventListener('pointerdown', onInteract);
   122→    el.addEventListener('wheel', onInteract, { passive: true });
   123→    return () => {
   124→      el.removeEventListener('pointerdown', onInteract);
   125→      el.removeEventListener('wheel', onInteract);
   126→    };
   127→  }, [gl]);
   128→
   129→  useEffect(() => {
   130→    modeSince.current = performance.now();
   131→    if (mode === 'overview') desired.current = { target: new THREE.Vector3(0, 0, 0), dist: 31, dir: new THREE.Vector3(0, 0.33, 0.94) };
   132→    else if (mode === 'nucleus') desired.current = { target: new THREE.Vector3(0, 0, 0), dist: 3.4, dir: new THREE.Vector3(0.35, 0.25, 0.9) };
   133→    else if (mode === 'membrane') {
   134→      const rec = bestReceptor(layout);
   135→      if (rec) desired.current = { target: toWorld(rec.pos), dist: 6.2, dir: new THREE.Vector3(0.15, 0.28, 0.94) };
   136→    }
   137→    // follow 每帧动态计算（不重置观察方位，尊重用户视角）
   138→  }, [mode, layout, spec]);
   139→
   140→  useFrame(() => {
   141→    const controls = controlsRef.current;
   142→    if (!controls) return;
   143→
   144→    const now = performance.now();
   145→    const userRecently = now - lastUser.current < 4000; // 用户 4s 内交互过 → 让位
   146→    const transitional = now - modeSince.current < 2000; // 模式切换 2s 过渡期
   147→    const following = (mode === 'follow' || mode === 'tour') && !userRecently;
   148→
   149→    // 稳态且无跟随任务: 完全交给 OrbitControls（用户自由旋转/缩放/autoRotate）
   150→    if (!transitional && !following) return;
   151→    // 过渡期内用户开始交互 → 立即让位
   152→    if (transitional && userRecently) return;
   153→
   154→    if (mode === 'follow') {
   155→      const target = trackedMolecule(layout);
   156→      if (target) {
   157→        desired.current.target.copy(toWorld(target.pos));
   158→        desired.current.dist = 6.5;
   159→        lastTracked.current = target.id;
   160→      }
   161→    } else if (mode === 'tour' && tourTarget) {
   162→      desired.current.target.copy(toWorld(tourTarget));
   163→      desired.current.dist = 7.8;
   164→    }
   165→
   166→    const dt = desired.current;
   167→    smoothTarget.current.lerp(dt.target, 0.045);
   168→    controls.target.copy(smoothTarget.current);
   169→
   170→    // 观察方位阻尼（预设机位过渡期恢复标准方位；follow/tour 保留用户视角）
   171→    const dir = camera.position.clone().sub(controls.target).normalize();
   172→    if (dt.dir && mode !== 'follow' && mode !== 'tour') {
   173→      dir.lerp(dt.dir, 0.06).normalize();
   174→    }
   175→
   176→    // 距离阻尼（沿当前方向推拉）
   177→    const d = camera.position.distanceTo(controls.target);
   178→    const nd = THREE.MathUtils.lerp(d, dt.dist, mode === 'follow' ? 0.05 : 0.06);
   179→    camera.position.copy(controls.target).add(dir.multiplyScalar(Math.max(1.2, nd)));
   180→    controls.update();
   181→  });
   182→
   183→  return null;
   184→}

   186→/** 活性最高的受体（质膜近景机位锚点） */
   187→function bestReceptor(layout: ReturnType<typeof layout3D> | null) {
   188→  if (!layout) return null;
   189→  const states = useLabStore.getState().nodeStates;
   190→  const receptors = layout.nodes.filter((n) => n.tier === 1);
   191→  if (!receptors.length) return null;
   192→  let best = receptors[0];
   193→  let bestA = -1;
   194→  for (const r of receptors) {
   195→    const a = states[r.id]?.activity ?? 0;
   196→    if (a > bestA) {
   197→      bestA = a;
   198→      best = r;
   199→    }
   200→  }
   201→  return best;
   202→}
   203→
   204→/** 最近激活且仍活跃的分子（跟随信号目标） */
   205→function trackedMolecule(layout: ReturnType<typeof layout3D> | null) {
   206→  if (!layout) return null;
   207→  const { events, nodeStates } = useLabStore.getState();
   208→  const byId = new Map(layout.nodes.map((n) => [n.id, n]));
   209→  for (let i = events.length - 1; i >= 0 && i > events.length - 12; i--) {
   210→    const ev = events[i];
   211→    if (!ev.nodeId) continue;
   212→    const st = nodeStates[ev.nodeId];
   213→    const node = byId.get(ev.nodeId);
   214→    if (st && node && st.activity > 0.35) return node;
   215→  }
   216→  return null;
   217→}

   436→interface LangCtx {
   437→  lang: Lang;
   438→  setLang: (l: Lang) => void;
   439→  t: (key: string) => string;
   440→}
   441→
   442→const LangContext = createContext<LangCtx>({
   443→  lang: 'zh',
   444→  setLang: () => {},
   445→  t: (k) => k,
   446→});
   447→
   448→const STORAGE_KEY = 'vcl-lang';
   449→const COOKIE_KEY = 'vcl-lang';
   450→const COOKIE_RE = new RegExp(`(?:^|;\\s*)${COOKIE_KEY}=([^;]*)`);
   451→
   452→/* 外部存储订阅（useSyncExternalStore 标准范式）: setLang 写入后 emit 通知全部消费组件 */
   453→const listeners = new Set<() => void>();
   454→function subscribe(cb: () => void) {
   455→  listeners.add(cb);
   456→  return () => {
   457→    listeners.delete(cb);
   458→  };
   459→}
   460→function emit() {
   461→  listeners.forEach((l) => l());
   462→}
   463→
   464→/** 客户端快照: cookie 优先（新版主存储）→ localStorage 兑底（旧版迁移读取路径） */
   465→function readClientLang(): Lang {
   466→  try {
   467→    const c = document.cookie.match(COOKIE_RE)?.[1];
   468→    if (c === 'en' || c === 'zh') return c;
   469→    const s = window.localStorage.getItem(STORAGE_KEY);
   470→    if (s === 'en' || s === 'zh') return s;
   471→  } catch {
   472→    /* localStorage 不可用（隐私模式） */
   473→  }
   474→  return 'zh';
   475→}
   476→
   477→/** 写 cookie（1 年）: 服务端与客户端共读同一份数据 → 刷新后无闪烁 */
   478→function writeCookie(l: Lang) {
   479→  try {
   480→    document.cookie = `${COOKIE_KEY}=${l}; path=/; max-age=31536000; samesite=lax`;
   481→  } catch {
   482→    /* ignore */
   483→  }
   484→}
   485→
   486→/** 切换语言（模块级: 纯外部存储写入 + 通知, 不依赖组件生命周期） */
   487→function setLang(l: Lang) {
   488→  writeCookie(l);
   489→  try {
   490→    window.localStorage.setItem(STORAGE_KEY, l);
   491→  } catch {
   492→    /* ignore */
   493→  }
   494→  emit();
   495→}
   496→
   497→export function LangProvider({ children, initialLang = 'zh' }: { children: ReactNode; initialLang?: Lang }) {
   498→  // useSyncExternalStore 水合安全范式:
   499→  //   水合期返回 server 快照（= 布局传入的 cookie 值）→ 与 SSR HTML 完全一致;
   500→  //   水合完成后切 client 快照（cookie/localStorage）→ 差异由 React 安全重渲染（无水合错配）
   501→  const lang = useSyncExternalStore(subscribe, readClientLang, () => initialLang);
   502→
   503→  // 挂载后一次性迁移: 旧版仅存 localStorage 的用户把语言提升写入 cookie（此后服务端直读, 无闪烁）
   504→  useEffect(() => {
   505→    try {
   506→      const hasCookie = COOKIE_RE.test(document.cookie);
   507→      const saved = window.localStorage.getItem(STORAGE_KEY);
   508→      if (!hasCookie && (saved === 'en' || saved === 'zh')) writeCookie(saved);
   509→    } catch {
   510→      /* ignore */
   511→    }
   512→  }, []);
   513→
   514→  // 语言切换时同步 <html lang>（无障碍 / 屏幕阅读器发音正确）
   515→  useEffect(() => {
   516→    document.documentElement.lang = lang === 'en' ? 'en' : 'zh-CN';
   517→  }, [lang]);
   518→
   519→  const value = useMemo<LangCtx>(
   520→    () => ({ lang, setLang, t: (k: string) => T[k]?.[lang] ?? k }),
   521→    [lang],
   522→  );
   523→
   524→  return <LangContext.Provider value={value}>{children}</LangContext.Provider>;
   525→}
   526→
   527→export function useLang(): LangCtx {
   528→  return useContext(LangContext);
   529→}

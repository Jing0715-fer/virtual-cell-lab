    34→function detectLowEndGpu(): boolean {
    35→  if (typeof navigator === 'undefined') return false;
    36→  const nav = navigator as Navigator & { deviceMemory?: number };
    37→  if (nav.deviceMemory !== undefined && nav.deviceMemory <= 4) return true;
    38→  if (nav.hardwareConcurrency && nav.hardwareConcurrency <= 4) return true;
    39→  try {
    40→    const c = document.createElement('canvas');
    41→    const gl = (c.getContext('webgl2') ?? c.getContext('webgl')) as WebGLRenderingContext | null;
    42→    if (!gl) return true;
    43→    const ext = gl.getExtension('WEBGL_debug_renderer_info');
    44→    const renderer = ext ? String(gl.getParameter(ext.UNMASKED_RENDERER_WEBGL)) : '';
    45→    return /swiftshader|llvmpipe|software|basic\s*render|angle \(.*software/i.test(renderer);
    46→  } catch {
    47→    return false;
    48→  }

    51→/** 切面控制器: 全局裁剪平面剖切细胞前半部，露出内部细胞器与核内分子（深度可调） */
    52→/* eslint-disable react-hooks/immutability -- renderer.clippingPlanes 为 three.js 全局渲染器命令式 API（R3F 标准用法） */
    53→function SectionClipController({ enabled, ringR, depth }: { enabled: boolean; ringR: number; depth: number }) {
    54→  const gl = useThree((s) => s.gl);
    55→  const scene = useThree((s) => s.scene);
    56→  // 剖面法向: 稍微俯视的前向切割（与默认相机方位一致，翻开“细胞剖面”）
    57→  const plane = useMemo(
    58→    () => new THREE.Plane(new THREE.Vector3(0, -0.22, -1).normalize(), 0.55),
    59→    [],
    60→  );
    61→  const origSides = useRef<Map<THREE.Material, THREE.Side>>(new Map());
    62→  const ringRef = useRef<THREE.Group | null>(null);
    63→
    64→  // 剖面深度（0=刚触及表面 1=深剖近后半）：平面常数 10 → -4（默认 0.65 ≈ 0.9，接近原固定值）
    65→  useEffect(() => {
    66→    plane.constant = 10 - depth * 14;
    67→    if (ringRef.current) {
    68→      ringRef.current.position.copy(plane.normal.clone().multiplyScalar(-plane.constant));
    69→    }
    70→  }, [depth, plane, enabled]);
    71→
    72→  useEffect(() => {
    73→    const restore = () => {
    74→      origSides.current.forEach((side, m) => {
    75→        m.side = side;
    76→        m.needsUpdate = true;
    77→      });
    78→      origSides.current.clear();
    79→    };
    80→    if (enabled) {
    81→      gl.clippingPlanes = [plane];
    82→      // 剖开后内壁可见: 结构材质临时双面化（记忆原 side 以便还原）
    83→      scene.traverse((obj) => {
    84→        const mesh = obj as THREE.Mesh;
    85→        if (!mesh.material) return;
    86→        const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
    87→        for (const m of mats) {
    88→          if (!(m instanceof THREE.Material)) continue;
    89→          if (!origSides.current.has(m)) origSides.current.set(m, m.side);
    90→          m.side = THREE.DoubleSide;
    91→          m.needsUpdate = true;
    92→        }
    93→      });
    94→    } else {
    95→      gl.clippingPlanes = [];
    96→    }
    97→    return () => {
    98→      gl.clippingPlanes = [];
    99→      restore();
   100→    };
   101→  }, [enabled, gl, scene, plane]);
   102→
   103→  // 剖面方位环（淡淡的两圈标记，指示切割平面位置与朝向）
   104→  const ringQuat = useMemo(() => {
   105→    const q = new THREE.Quaternion();
   106→    q.setFromUnitVectors(new THREE.Vector3(0, 0, 1), plane.normal);
   107→    return q;
   108→  }, [plane]);
   109→
   110→  return enabled ? (
   111→    <group ref={ringRef} quaternion={ringQuat}>
   112→      <mesh>
   113→        <ringGeometry args={[ringR - 0.12, ringR, 96]} />
   114→        <meshBasicMaterial color="#5eead4" transparent opacity={0.22} side={THREE.DoubleSide} fog={false} />
   115→      </mesh>
   116→      <mesh>
   117→        <ringGeometry args={[ringR * 0.42, ringR * 0.42 + 0.05, 64]} />
   118→        <meshBasicMaterial color="#5eead4" transparent opacity={0.1} side={THREE.DoubleSide} fog={false} />
   119→      </mesh>
   120→    </group>
   121→  ) : null;
   122→}

   124→/** 相机驱动器: 预设机位（含标准观察方位） + 信号跟随 + 教学聚焦（阻尼插值） */
   125→function CameraRig({ mode, layout, spec, controlsRef, tourTarget }: {
   126→  mode: CamMode;
   127→  layout: ReturnType<typeof layout3D> | null;
   128→  spec: { scale: [number, number, number] };
   129→  controlsRef: RefObject<OrbitControlsImpl | null>;
   130→  /** 教学引导: 当前聚焦分子世界坐标 */
   131→  tourTarget: Vec3 | null;
   132→}) {
   133→  const { camera } = useThree();
   134→  const desired = useRef({ target: new THREE.Vector3(), dist: 30, dir: new THREE.Vector3(0, 0.33, 0.94) });
   135→  const smoothTarget = useRef(new THREE.Vector3());
   136→  const lastTracked = useRef<string | null>(null);
   137→
   138→  const toWorld = (p: Vec3): THREE.Vector3 =>
   139→    new THREE.Vector3(p.x * spec.scale[0], p.y * spec.scale[1], p.z * spec.scale[2]);
   140→
   141→  useEffect(() => {
   142→    if (mode === 'overview') desired.current = { target: new THREE.Vector3(0, 0, 0), dist: 31, dir: new THREE.Vector3(0, 0.33, 0.94) };
   143→    else if (mode === 'nucleus') desired.current = { target: new THREE.Vector3(0, 0, 0), dist: 3.4, dir: new THREE.Vector3(0.35, 0.25, 0.9) };
   144→    else if (mode === 'membrane') {
   145→      const rec = bestReceptor(layout);
   146→      if (rec) desired.current = { target: toWorld(rec.pos), dist: 6.2, dir: new THREE.Vector3(0.15, 0.28, 0.94) };
   147→    }
   148→    // follow 每帧动态计算（不重置观察方位，尊重用户视角）
   149→  }, [mode, layout, spec]);
   150→
   151→  useFrame(() => {
   152→    const controls = controlsRef.current;
   153→    if (!controls) return;
   154→
   155→    if (mode === 'follow') {
   156→      const target = trackedMolecule(layout);
   157→      if (target) {
   158→        desired.current.target.copy(toWorld(target.pos));
   159→        desired.current.dist = 6.5;
   160→        lastTracked.current = target.id;
   161→      }
   162→    } else if (mode === 'tour' && tourTarget) {
   163→      desired.current.target.copy(toWorld(tourTarget));
   164→      desired.current.dist = 7.8;
   165→    }
   166→
   167→    const dt = desired.current;
   168→    smoothTarget.current.lerp(dt.target, 0.045);
   169→    controls.target.copy(smoothTarget.current);
   170→
   171→    // 观察方位阻尼（预设机位恢复标准方位；follow/free/tour 保留用户视角）
   172→    const dir = camera.position.clone().sub(controls.target).normalize();
   173→    if (dt.dir && mode !== 'follow' && mode !== 'free' && mode !== 'tour') {
   174→      dir.lerp(dt.dir, 0.06).normalize();
   175→    }
   176→
   177→    // 距离阻尼（沿当前方向推拉）
   178→    const d = camera.position.distanceTo(controls.target);
   179→    const nd = THREE.MathUtils.lerp(d, dt.dist, mode === 'follow' ? 0.05 : 0.06);
   180→    camera.position.copy(controls.target).add(dir.multiplyScalar(Math.max(1.2, nd)));
   181→    controls.update();
   182→  });
   183→
   184→  return null;
   185→}

   187→/** 活性最高的受体（质膜近景机位锚点） */
   188→function bestReceptor(layout: ReturnType<typeof layout3D> | null) {
   189→  if (!layout) return null;
   190→  const states = useLabStore.getState().nodeStates;
   191→  const receptors = layout.nodes.filter((n) => n.tier === 1);
   192→  if (!receptors.length) return null;
   193→  let best = receptors[0];
   194→  let bestA = -1;
   195→  for (const r of receptors) {
   196→    const a = states[r.id]?.activity ?? 0;
   197→    if (a > bestA) {
   198→      bestA = a;
   199→      best = r;
   200→    }
   201→  }
   202→  return best;
   203→}

   205→/** 最近激活且仍活跃的分子（跟随信号目标） */
   206→function trackedMolecule(layout: ReturnType<typeof layout3D> | null) {
   207→  if (!layout) return null;
   208→  const { events, nodeStates } = useLabStore.getState();
   209→  const byId = new Map(layout.nodes.map((n) => [n.id, n]));
   210→  for (let i = events.length - 1; i >= 0 && i > events.length - 12; i--) {
   211→    const ev = events[i];
   212→    if (!ev.nodeId) continue;
   213→    const st = nodeStates[ev.nodeId];
   214→    const node = byId.get(ev.nodeId);
   215→    if (st && node && st.activity > 0.35) return node;
   216→  }
   217→  return null;
   218→}
